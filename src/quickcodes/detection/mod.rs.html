92impl AdvancedDetector {
93    /// Create new detector with configuration
94    pub fn new(config: DetectionConfig) -> Self {
95        Self { config }
96    }
97
98    /// Create detector with default configuration
99    pub fn default() -> Self {
100        Self::new(DetectionConfig::default())
101    }
102
103    /// Detect all barcodes in image with advanced algorithms
104    pub fn detect_all(&self, image: &ImageBuffer<Luma<u8>, Vec<u8>>) -> Vec<AdvancedDetectionResult> {
105        // Multi-stage detection pipeline
106        
107        // Stage 1: Preprocessing
108        let processed_images = preprocessing::preprocess_image(image, &self.config);
109        
110        // Stage 2: Pattern detection per format
111        let mut all_candidates = Vec::new();
112        
113        for processed_img in &processed_images {
114            // Try each target format
115            for format in &self.config.target_formats {
116                let candidates = match format {
117                    BarcodeType::QRCode => engines::qr::detect_qr_candidates(processed_img),
118                    BarcodeType::DataMatrix => engines::datamatrix::detect_datamatrix_candidates(processed_img),
119                    BarcodeType::PDF417 => engines::pdf417::detect_pdf417_candidates(processed_img),
120                    BarcodeType::Aztec => engines::aztec::detect_aztec_candidates(processed_img),
121                    _ => engines::linear::detect_linear_candidates(processed_img, format),
122                };
123                all_candidates.extend(candidates);
124            }
125        }
126        
127        // Stage 3: Validation and scoring
128        let mut validated_results = Vec::new();
129        for candidate in all_candidates {
130            if let Some(result) = validation::validate_and_score(candidate, &self.config) {
131                if result.confidence >= self.config.min_confidence {
132                    validated_results.push(result);
133                }
134            }
135        }
136        
137        // Stage 4: Non-maximum suppression and final filtering
138        validation::non_maximum_suppression(validated_results, &self.config)
139    }
140
141    /// Detect first barcode in image
142    pub fn detect_first(&self, image: &ImageBuffer<Luma<u8>, Vec<u8>>) -> Option<AdvancedDetectionResult> {
143        let mut config = self.config.clone();
144        config.max_codes_per_image = 1;
145        
146        let detector = AdvancedDetector::new(config);
147        detector.detect_all(image).into_iter().next()
148    }
149}

160/// Pattern-specific data for validation
161#[derive(Debug, Clone)]
162pub enum PatternData {
163    QRCode {
164        finder_patterns: Vec<FinderPattern>,
165        alignment_patterns: Vec<(f32, f32)>,
166        timing_patterns: (Vec<bool>, Vec<bool>),
167    },
168    DataMatrix {
169        l_border: LBorder,
170        timing_patterns: (Vec<bool>, Vec<bool>), // (vertical, horizontal)
171        module_size: f32,
172    },
173    Linear {
174        bars_and_spaces: Vec<u32>,
175        start_end_patterns: (Vec<u32>, Vec<u32>),
176        scan_line: (u32, u32, u32, u32), // x1, y1, x2, y2
177    },
178    PDF417 {
179        start_patterns: Vec<(u32, u32)>,
180        stop_patterns: Vec<(u32, u32)>,
181        rows: Vec<Vec<u32>>,
182    },
183    Aztec {
184        bullseye: (f32, f32, f32), // x, y, size
185        reference_grid: Vec<Vec<bool>>,
186    },
187}
