11impl ValidationPipeline {
12    /// Complete validation and scoring pipeline
13    pub fn validate_and_score(
14        candidate: DetectionCandidate,
15        config: &DetectionConfig,
16    ) -> Option<AdvancedDetectionResult> {
17        // Stage 1: Quick rejection filters
18        if !Self::pre_filter_candidate(&candidate) {
19            return None;
20        }
21        
22        // Stage 2: Comprehensive confidence scoring
23        let confidence_score = ConfidenceScorer::calculate_confidence(&candidate);
24        
25        // Stage 3: Threshold filtering
26        if confidence_score.overall < config.min_confidence {
27            return None;
28        }
29        
30        // Stage 4: Content validation (try to decode)
31        let validated_data = Self::validate_content(&candidate, &confidence_score)?;
32        
33        // Stage 5: Final contextual validation
34        if !Self::final_validation(&candidate, &confidence_score, config) {
35            return None;
36        }
37        
38        Some(AdvancedDetectionResult {
39            data: validated_data,
40            barcode_type: candidate.barcode_type,
41            confidence: confidence_score.overall,
42            geometric_score: confidence_score.geometric,
43            pattern_score: confidence_score.pattern,
44            content_score: confidence_score.content,
45            position: candidate.position,
46        })
47    }

49    /// Stage 1: Quick pre-filtering to reject obvious false positives
50    fn pre_filter_candidate(candidate: &DetectionCandidate) -> bool {
51        // Basic sanity checks
52        if candidate.position.width == 0 || candidate.position.height == 0 {
53            return false;
54        }
55        
56        // Size constraints
57        if candidate.position.width < 10 || candidate.position.height < 10 {
58            return false;
59        }
60        
61        if candidate.position.width > 5000 || candidate.position.height > 5000 {
62            return false;
63        }
64        
65        // Aspect ratio constraints based on barcode type
66        let aspect_ratio = candidate.position.width as f32 / candidate.position.height as f32;
67        
68        match candidate.barcode_type {
69            // 2D codes should be roughly square
70            crate::types::BarcodeType::QRCode | 
71            crate::types::BarcodeType::DataMatrix | 
72            crate::types::BarcodeType::Aztec => {
73                if aspect_ratio < 0.3 || aspect_ratio > 3.0 {
74                    return false;
75                }
76            },
77            // PDF417 can be more rectangular
78            crate::types::BarcodeType::PDF417 => {
79                if aspect_ratio < 0.1 || aspect_ratio > 10.0 {
80                    return false;
81                }
82            },
83            // 1D codes should be wider than tall
84            _ => {
85                if aspect_ratio < 1.5 || aspect_ratio > 20.0 {
86                    return false;
87                }
88            }
89        }
90        
91        true
92    }

94    /// Stage 4: Content validation through decoding attempt
95    fn validate_content(
96        candidate: &DetectionCandidate, 
97        confidence_score: &ConfidenceScore
98    ) -> Option<String> {
99        // If we already have extracted data with good content score, use it
100        if let Some(ref data) = candidate.raw_data {
101            if confidence_score.content > 0.7 {
102                return Some(data.clone());
103            }
104        }
105        
106        // Otherwise, attempt decoding based on pattern data
107        match &candidate.pattern_data {
108            crate::detection::PatternData::QRCode { .. } => {
109                // Use existing QR decoder
110                Self::decode_qr_from_pattern(candidate)
111            },
112            crate::detection::PatternData::DataMatrix { .. } => {
113                // Use existing DataMatrix decoder
114                Self::decode_datamatrix_from_pattern(candidate)
115            },
116            crate::detection::PatternData::Linear { .. } => {
117                // Use existing linear decoders
118                Self::decode_linear_from_pattern(candidate)
119            },
120            crate::detection::PatternData::PDF417 { .. } => {
121                // Use existing PDF417 decoder
122                Self::decode_pdf417_from_pattern(candidate)
123            },
124            crate::detection::PatternData::Aztec { .. } => {
125                // Use existing Aztec decoder
126                Self::decode_aztec_from_pattern(candidate)
127            },
128        }
129    }

131    /// Stage 5: Final contextual validation
132    fn final_validation(
133        candidate: &DetectionCandidate,
134        confidence_score: &ConfidenceScore,
135        _config: &DetectionConfig,
136    ) -> bool {
137        // Ensure minimum scores in critical areas
138        if confidence_score.geometric < 0.5 {
139            return false;
140        }
141        
142        if confidence_score.pattern < 0.4 {
143            return false;
144        }
145        
146        // Format-specific validation
147        match candidate.barcode_type {
148            crate::types::BarcodeType::QRCode => {
149                // QR codes need high pattern integrity
150                confidence_score.pattern >= 0.6
151            },
152            crate::types::BarcodeType::DataMatrix => {
153                // DataMatrix needs good geometric score
154                confidence_score.geometric >= 0.6
155            },
156            crate::types::BarcodeType::EAN13 | 
157            crate::types::BarcodeType::UPCA |
158            crate::types::BarcodeType::ITF14 => {
159                // These need valid checksums
160                confidence_score.content >= 0.9
161            },
162            _ => true,
163        }
164    }

166    /// Non-maximum suppression to eliminate duplicate detections
167    pub fn non_maximum_suppression(
168        mut results: Vec<AdvancedDetectionResult>,
169        config: &DetectionConfig,
170    ) -> Vec<AdvancedDetectionResult> {
171        if results.is_empty() {
172            return results;
173        }
174        
175        // Sort by confidence score (highest first)
176        results.sort_by(|a, b| b.confidence.partial_cmp(&a.confidence).unwrap());
177        
178        let mut final_results = Vec::new();
179        let mut suppressed = vec![false; results.len()];
180        
181        for i in 0..results.len() {
182            if suppressed[i] {
183                continue;
184            }
185            
186            final_results.push(results[i].clone());
187            
188            // Suppress overlapping detections with lower confidence
189            for j in (i+1)..results.len() {
190                if suppressed[j] {
191                    continue;
192                }
193                
194                let overlap = Self::calculate_overlap(&results[i].position, &results[j].position);
195                
196                // Suppress if significant overlap (>50%) and same type
197                if overlap > 0.5 && results[i].barcode_type == results[j].barcode_type {
198                    suppressed[j] = true;
199                }
200                
201                // Also suppress if very high overlap regardless of type
202                if overlap > 0.8 {
203                    suppressed[j] = true;
204                }
205            }
206            
207            // Respect max codes limit
208            if final_results.len() >= config.max_codes_per_image {
209                break;
210            }
211        }
212        
213        final_results
214    }

216    /// Calculate overlap ratio between two bounding boxes
217    fn calculate_overlap(box1: &BoundingBox, box2: &BoundingBox) -> f32 {
218        let x1 = box1.x.max(box2.x);
219        let y1 = box1.y.max(box2.y);
220        let x2 = (box1.x + box1.width).min(box2.x + box2.width);
221        let y2 = (box1.y + box1.height).min(box2.y + box2.height);
222        
223        if x2 <= x1 || y2 <= y1 {
224            return 0.0; // No overlap
225        }
226        
227        let intersection = (x2 - x1) * (y2 - y1);
228        let area1 = box1.width * box1.height;
229        let area2 = box2.width * box2.height;
230        let union = area1 + area2 - intersection;
231        
232        if union == 0 {
233            return 0.0;
234        }
235        
236        intersection as f32 / union as f32
237    }

239    /// Contextual analysis for adaptive thresholding
240    pub fn analyze_detection_context(
241        results: &[AdvancedDetectionResult],
242        image_quality: &ImageQualityMetrics,
243    ) -> DetectionContext {
244        let mut context = DetectionContext::default();
245        
246        // Analyze detection density
247        context.detection_density = results.len() as f32 / 100.0; // Normalized per 100 codes
248        
249        // Analyze confidence distribution
250        if !results.is_empty() {
251            let total_confidence: f32 = results.iter().map(|r| r.confidence).sum();
252            context.avg_confidence = total_confidence / results.len() as f32;
253            
254            let max_confidence = results.iter()
255                .map(|r| r.confidence)
256                .fold(0.0f32, f32::max);
257            context.max_confidence = max_confidence;
258        }
259        
260        // Use image quality metrics
261        context.image_noise_level = image_quality.noise_level;
262        context.image_contrast = image_quality.overall_contrast;
263        context.image_sharpness = image_quality.sharpness_score;
264        
265        // Suggest adaptive threshold
266        context.suggested_threshold = Self::calculate_adaptive_threshold(&context);
267        
268        context
269    }

271    /// Calculate adaptive threshold based on context
272    fn calculate_adaptive_threshold(context: &DetectionContext) -> f32 {
273        let mut threshold: f32 = 0.85; // Base threshold
274        
275        // Adjust for image quality
276        if context.image_noise_level > 0.5 {
277            threshold += 0.05; // Be more strict with noisy images
278        }
279        
280        if context.image_contrast < 0.3 {
281            threshold += 0.03; // Be more strict with low contrast
282        }
283        
284        if context.image_sharpness < 0.3 {
285            threshold += 0.02; // Be more strict with blurry images
286        }
287        
288        // Adjust for detection patterns
289        if context.detection_density > 0.1 {
290            threshold += 0.02; // Be more strict when many detections
291        }
292        
293        if context.avg_confidence < 0.7 {
294            threshold -= 0.02; // Be less strict if overall confidence is low
295        }
296        
297        threshold.min(0.95).max(0.75) // Clamp to reasonable range
298    }
