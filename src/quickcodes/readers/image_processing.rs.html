23impl Region {
24    #[allow(dead_code)]
25    pub fn new(x: u32, y: u32, width: u32, height: u32) -> Self {
26        Self { x, y, width, height }
27    }
28
29    pub fn extract(&self, image: &GrayImage) -> Result<GrayImage> {
30        let mut region_image = GrayImage::new(self.width, self.height);
31        
32        for y in 0..self.height {
33            for x in 0..self.width {
34                let src_x = self.x + x;
35                let src_y = self.y + y;
36                
37                if src_x < image.width() && src_y < image.height() {
38                    let pixel = image.get_pixel(src_x, src_y);
39                    region_image.put_pixel(x, y, *pixel);
40                }
41            }
42        }
43        
44        Ok(region_image)
45    }
46}

48/// Prepara a imagem para detecção de códigos
49pub(crate) fn prepare_image(image: &DynamicImage) -> Result<GrayImage> {
50    // Converter para escala de cinza
51    let gray = image.to_luma8();
52
53    #[cfg(feature = "readers")]
54    {
55        // Aplicar blur gaussiano para reduzir ruído
56        let blurred = gaussian_blur_f32(&gray, 1.0);
57
58        // Binarização usando threshold simples
59        let binary = threshold(&blurred, 128);
60
61        Ok(binary)
62    }
63
64    #[cfg(not(feature = "readers"))]
65    {
66        Ok(gray)
67    }
68}

70/// Corrige a orientação da imagem detectando rotação automática
71#[cfg(feature = "readers")]
72pub(crate) fn correct_orientation(image: &GrayImage) -> Result<GrayImage> {
73    // Detectar rotação baseada em linhas dominantes
74    let angle = detect_rotation_angle(image)?;
75    
76    if angle.abs() > 1.0 {
77        // Aplicar rotação se o ângulo for significativo (> 1 grau)
78        let rotated = rotate_about_center(image, angle, Interpolation::Bilinear, Luma([255]));
79        Ok(rotated)
80    } else {
81        Ok(image.clone())
82    }
83}

85/// Detecta o ângulo de rotação da imagem analisando linhas dominantes
86#[cfg(feature = "readers")]
87fn detect_rotation_angle(image: &GrayImage) -> Result<f32> {
88    // Aplicar filtro de detecção de bordas
89    let edges = canny(image, 50.0, 100.0);
90    
91    // Detectar linhas usando transformada de Hough
92    let options = LineDetectionOptions {
93        vote_threshold: 40,
94        suppression_radius: 8,
95    };
96    
97    let lines = detect_lines(&edges, options);
98    
99    if lines.is_empty() {
100        return Ok(0.0);
101    }
102    
103    // Analisar ângulos das linhas para encontrar orientação dominante
104    let mut angles = Vec::new();
105    
106    for line in &lines {
107        // Converter ângulo polar para graus
108        let angle_deg = line.angle_in_degrees as f32;
109        
110        // Normalizar ângulo para [-45, 45] graus
111        let normalized_angle = normalize_angle_for_correction(angle_deg);
112        angles.push(normalized_angle);
113    }
114    
115    if angles.is_empty() {
116        return Ok(0.0);
117    }
118    
119    // Encontrar ângulo mais comum (moda aproximada)
120    angles.sort_by(|a, b| a.partial_cmp(b).unwrap());
121    
122    // Usar mediana como ângulo de correção
123    let median_angle = if angles.len() % 2 == 0 {
124        (angles[angles.len() / 2 - 1] + angles[angles.len() / 2]) / 2.0
125    } else {
126        angles[angles.len() / 2]
127    };
128    
129    Ok(-median_angle) // Negativo para corrigir a rotação
130}

132/// Normaliza ângulo para correção de rotação
133fn normalize_angle_for_correction(angle: f32) -> f32 {
134    let mut normalized = angle % 180.0;
135    
136    // Converter para range [-90, 90]
137    if normalized > 90.0 {
138        normalized -= 180.0;
139    } else if normalized < -90.0 {
140        normalized += 180.0;
141    }
142    
143    // Limitar a [-45, 45] para correções menores
144    if normalized > 45.0 {
145        normalized -= 90.0;
146    } else if normalized < -45.0 {
147        normalized += 90.0;
148    }
149    
150    normalized
151}

153/// Corrige distorção de perspectiva detectando cantos de códigos retangulares
154#[cfg(feature = "readers")]
155pub(crate) fn correct_perspective(image: &GrayImage) -> Result<GrayImage> {
156    // Detectar cantos candidatos
157    let corners = detect_barcode_corners(image)?;
158    
159    if corners.len() >= 4 {
160        // Ordenar cantos para formar retângulo
161        let ordered_corners = order_corners_clockwise(&corners);
162        
163        if let Some(corners) = ordered_corners {
164            // Calcular transformação de perspectiva
165            return apply_perspective_correction(image, &corners);
166        }
167    }
168    
169    // Se não conseguir detectar perspectiva, retornar imagem original
170    Ok(image.clone())
171}

173/// Detecta cantos que podem pertencer a códigos de barras
174#[cfg(feature = "readers")]
175fn detect_barcode_corners(image: &GrayImage) -> Result<Vec<Corner>> {
176    // Aplicar detecção de cantos FAST
177    let corners = corners_fast9(image, 12);
178    
179    // Filtrar cantos baseado em contexto local
180    let mut filtered_corners = Vec::new();
181    
182    for corner in corners {
183        if is_corner_in_barcode_context(image, &corner) {
184            filtered_corners.push(corner);
185        }
186    }
187    
188    Ok(filtered_corners)
189}

191/// Verifica se um canto está em contexto de código de barras
192fn is_corner_in_barcode_context(image: &GrayImage, corner: &Corner) -> bool {
193    let x = corner.x as u32;
194    let y = corner.y as u32;
195    
196    // Verificar se está dentro dos limites
197    if x < 10 || y < 10 || x >= image.width() - 10 || y >= image.height() - 10 {
198        return false;
199    }
200    
201    // Analisar região local ao redor do canto
202    let region_size = 20;
203    let mut light_pixels = 0;
204    let mut dark_pixels = 0;
205    
206    for dy in 0..region_size {
207        for dx in 0..region_size {
208            let px = x + dx - region_size / 2;
209            let py = y + dy - region_size / 2;
210            
211            if px < image.width() && py < image.height() {
212                let pixel = image.get_pixel(px, py)[0];
213                if pixel > 128 {
214                    light_pixels += 1;
215                } else {
216                    dark_pixels += 1;
217                }
218            }
219        }
220    }
221    
222    // Considerar como candidato se há contraste significativo
223    let total_pixels = light_pixels + dark_pixels;
224    let contrast_ratio = (light_pixels as f32) / (total_pixels as f32);
225    
226    // Código de barras deve ter contraste entre 20% e 80%
227    contrast_ratio > 0.2 && contrast_ratio < 0.8
228}

230/// Ordena cantos em sentido horário para formar retângulo
231fn order_corners_clockwise(corners: &[Corner]) -> Option<[Corner; 4]> {
232    if corners.len() < 4 {
233        return None;
234    }
235    
236    // Encontrar centro aproximado
237    let center_x = corners.iter().map(|c| c.x as f32).sum::<f32>() / corners.len() as f32;
238    let center_y = corners.iter().map(|c| c.y as f32).sum::<f32>() / corners.len() as f32;
239    
240    // Classificar cantos por quadrante
241    let mut top_left: Option<Corner> = None;
242    let mut top_right: Option<Corner> = None;
243    let mut bottom_left: Option<Corner> = None;
244    let mut bottom_right: Option<Corner> = None;
245    
246    for corner in corners {
247        let is_left = (corner.x as f32) < center_x;
248        let is_top = (corner.y as f32) < center_y;
249        
250        match (is_left, is_top) {
251            (true, true) => {
252                if top_left.is_none() || corner.x + corner.y < top_left.unwrap().x + top_left.unwrap().y {
253                    top_left = Some(*corner);
254                }
255            }
256            (false, true) => {
257                if top_right.is_none() || (corner.x as i32 - corner.y as i32) > (top_right.unwrap().x as i32 - top_right.unwrap().y as i32) {
258                    top_right = Some(*corner);
259                }
260            }
261            (true, false) => {
262                if bottom_left.is_none() || (corner.y as i32 - corner.x as i32) > (bottom_left.unwrap().y as i32 - bottom_left.unwrap().x as i32) {
263                    bottom_left = Some(*corner);
264                }
265            }
266            (false, false) => {
267                if bottom_right.is_none() || corner.x + corner.y > bottom_right.unwrap().x + bottom_right.unwrap().y {
268                    bottom_right = Some(*corner);
269                }
270            }
271        }
272    }
273    
274    // Verificar se encontramos os 4 cantos
275    if let (Some(tl), Some(tr), Some(bl), Some(br)) = (top_left, top_right, bottom_left, bottom_right) {
276        Some([tl, tr, br, bl]) // Sentido horário
277    } else {
278        None
279    }
280}

282/// Aplica correção de perspectiva usando os 4 cantos detectados
283#[cfg(feature = "readers")]
284fn apply_perspective_correction(image: &GrayImage, corners: &[Corner; 4]) -> Result<GrayImage> {
285    // Definir dimensões do retângulo de saída
286    let output_width = 400;
287    let output_height = 400;
288    
289    // Pontos de destino (retângulo perfeito)
290    let dst_corners = [
291        (0.0, 0.0),                                          // top-left
292        (output_width as f32, 0.0),                         // top-right  
293        (output_width as f32, output_height as f32),        // bottom-right
294        (0.0, output_height as f32),                        // bottom-left
295    ];
296    
297    // Pontos de origem (cantos detectados)
298    let src_corners = [
299        (corners[0].x as f32, corners[0].y as f32),
300        (corners[1].x as f32, corners[1].y as f32),
301        (corners[2].x as f32, corners[2].y as f32), 
302        (corners[3].x as f32, corners[3].y as f32),
303    ];
304    
305    // Calcular transformação de perspectiva
306    if let Some(projection) = Projection::from_control_points(src_corners, dst_corners) {
307        let corrected = warp(image, &projection, Interpolation::Bilinear, Luma([255]));
308        Ok(corrected)
309    } else {
310        // Se não conseguir calcular projeção, retornar original
311        Ok(image.clone())
312    }
313}

315/// Detecta e corrige múltiplas orientações de códigos na mesma imagem
316#[cfg(feature = "readers")]
317pub(crate) fn detect_and_correct_multiple_orientations(image: &GrayImage) -> Result<Vec<GrayImage>> {
318    let mut corrected_images = Vec::new();
319    
320    // Imagem original
321    corrected_images.push(image.clone());
322    
323    // Rotações comuns: 90°, 180°, 270°
324    let rotation_angles = [90.0, 180.0, 270.0];
325    
326    for &angle in &rotation_angles {
327        let radians = angle * PI / 180.0;
328        let rotated = rotate_about_center(image, radians, Interpolation::Bilinear, Luma([255]));
329        corrected_images.push(rotated);
330    }
331    
332    // Correção de perspectiva na imagem original
333    let perspective_corrected = correct_perspective(image)?;
334    if !images_are_similar(image, &perspective_corrected) {
335        corrected_images.push(perspective_corrected);
336    }
337    
338    Ok(corrected_images)
339}

341/// Verifica se duas imagens são similares (para evitar duplicatas)
342fn images_are_similar(img1: &GrayImage, img2: &GrayImage) -> bool {
343    if img1.width() != img2.width() || img1.height() != img2.height() {
344        return false;
345    }
346    
347    let total_pixels = (img1.width() * img1.height()) as f32;
348    let mut different_pixels = 0;
349    
350    for y in 0..img1.height() {
351        for x in 0..img1.width() {
352            let pixel1 = img1.get_pixel(x, y)[0];
353            let pixel2 = img2.get_pixel(x, y)[0];
354            
355            if (pixel1 as i32 - pixel2 as i32).abs() > 10 {
356                different_pixels += 1;
357            }
358        }
359    }
360    
361    let difference_ratio = different_pixels as f32 / total_pixels;
362    difference_ratio < 0.05 // Menos de 5% de diferença
363}
